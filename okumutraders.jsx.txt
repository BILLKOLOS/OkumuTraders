const Y_MIN = -3.0;
const Y_MAX  =  3.0;
const N      = 90;
const Y_LABELS = [3.0, 1.5, 0.0, -1.5, -3.0];

function genPt(t) {
  return (
    Math.sin(t * 0.38) * 1.35 +
    Math.sin(t * 1.15) * 0.55 +
    Math.sin(t * 2.9)  * 0.28 +
    (Math.random() - 0.5) * 0.22
  );
}

function LiveChart({ data, rate }) {
  const W = 1000;
  const H = 260;
  const yRange = Y_MAX - Y_MIN;
  const toX = (i) => (i / (data.length - 1)) * W;
  const toY = (v) => ((Y_MAX - v) / yRange) * H;
  const zeroY = toY(0);

  if (data.length < 2) return null;

  const pts  = data.map((v, i) => ({ x: toX(i), y: toY(v) }));
  const last  = pts[pts.length - 1];
  const first = pts[0];

  const linePath = pts
    .map((p, i) => `${i === 0 ? "M" : "L"}${p.x.toFixed(1)},${p.y.toFixed(1)}`)
    .join(" ");

  const areaPath =
    `${linePath} ` +
    `L${last.x.toFixed(1)},${zeroY.toFixed(1)} ` +
    `L${first.x.toFixed(1)},${zeroY.toFixed(1)} Z`;

  const zY = +zeroY.toFixed(1);

  return (
    <svg width="100%" viewBox={`0 0 ${W} ${H}`} preserveAspectRatio="none" style={{ display: "block" }}>
      <defs>
        <clipPath id="ot-above">
          <rect x="0" y="0" width={W} height={zY} />
        </clipPath>
        <clipPath id="ot-below">
          <rect x="0" y={zY} width={W} height={H - zY} />
        </clipPath>
        <linearGradient id="ot-gg" x1="0" y1="0" x2="0" y2="1">
          <stop offset="0%"   stopColor="#1cd25e" stopOpacity="0.6" />
          <stop offset="100%" stopColor="#1cd25e" stopOpacity="0.04" />
        </linearGradient>
        <linearGradient id="ot-rg" x1="0" y1="0" x2="0" y2="1">
          <stop offset="0%"   stopColor="#ff2d55" stopOpacity="0.03" />
          <stop offset="100%" stopColor="#ff2d55" stopOpacity="0.5" />
        </linearGradient>
      </defs>

      {/* Zero baseline */}
      <line x1="0" y1={zY} x2={W} y2={zY} stroke="#253a58" strokeWidth="1.5" />

      {/* Green fill above zero */}
      <path d={areaPath} fill="url(#ot-gg)" clipPath="url(#ot-above)" />

      {/* Red fill below zero */}
      <path d={areaPath} fill="url(#ot-rg)" clipPath="url(#ot-below)" />

      {/* Price line */}
      <path
        d={linePath}
        fill="none"
        stroke="#1cd25e"
        strokeWidth="2.5"
        strokeLinejoin="round"
        strokeLinecap="round"
      />

      {/* Live dot + halo */}
      <circle cx={last.x} cy={last.y} r="7"  fill="#1cd25e" />
      <circle cx={last.x} cy={last.y} r="15" fill="#1cd25e" fillOpacity="0.18" />

      {/* Floating rate label */}
      <rect x={W - 190} y="10" width="180" height="36" rx="6" fill="#0f1f36" />
      <text
        x={W - 100} y="34"
        textAnchor="middle"
        fill="white"
        fontSize="18"
        fontFamily="'Courier New', monospace"
        fontWeight="800"
      >
        Rate: {rate.toFixed(4)}
      </text>
    </svg>
  );
}

        {/* AUTOSELL + P&L */}
        <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: 10 }}>
          <div style={{ display: "flex", alignItems: "center", gap: 8 }}>
            <div style={{
              width: 30, height: 30, borderRadius: "50%",
              background: `linear-gradient(135deg, ${C.green}, #0a8f3a)`,
              display: "flex", alignItems: "center", justifyContent: "center",
              fontSize: 12, fontWeight: 800, color: C.bg,
            }}>10</div>
            <span style={{ fontSize: 12, fontWeight: 700, color: C.muted, letterSpacing: 1 }}>AUTOSELL</span>
          </div>
          <div style={{ display: "flex", alignItems: "center", gap: 5 }}>
            <span style={{ fontSize: 11, fontWeight: 700, color: C.muted }}>P&L</span>
            <span style={{ fontWeight: 800, fontSize: 14, color: C.muted }}>KES</span>
            <span style={{ fontWeight: 900, fontSize: 16, color: pl > 0 ? C.green : pl < 0 ? C.red : C.muted }}>
              {pl > 0 ? "+" : ""}{pl.toFixed(2)}
            </span>
          </div>
        </div>

        {/* BUY / SELL */}
        <div style={{ display: "flex", gap: 10 }}>
          <button onClick={() => doTrade("BUY")} style={{
            flex: 1, padding: "16px 0",
            background: activeTrade?.type === "BUY" ? "#0a6e2a" : C.green,
            border: "none", borderRadius: 11,
            fontSize: 17, fontWeight: 900, color: C.bg, cursor: "pointer", letterSpacing: 1,
          }}>
            {activeTrade?.type === "BUY" ? "CLOSE" : "BUY"}
          </button>
          <button onClick={() => doTrade("SELL")} style={{
            flex: 1, padding: "16px 0",
            background: activeTrade?.type === "SELL" ? "#6e0a18" : C.red,
            border: "none", borderRadius: 11,
            fontSize: 17, fontWeight: 900, color: "#fff", cursor: "pointer", letterSpacing: 1,
          }}>
            {activeTrade?.type === "SELL" ? "CLOSE" : "SELL"}
          </button>
        </div>
      </div>

      <div style={{ flex: 1, minHeight: 0 }} />

      <BottomNav screen={screen} setScreen={setScreen} openDeposit={() => setShowDeposit(true)} />
      <Licence />

      {/* ── LOGIN MODAL ─────────────────────────────────────────────────────── */}
      {showLogin && (
        <div style={{
          position: "fixed", inset: 0, background: "rgba(0,0,0,0.8)",
          display: "flex", alignItems: "center", justifyContent: "center",
          zIndex: 100, padding: 20,
        }}>
          <div style={{
            background: C.card, borderRadius: 20, padding: "30px 22px",
            width: "100%", maxWidth: 360, border: `1.5px solid ${C.border}`,
          }}>
            <div style={{ display: "flex", alignItems: "center", gap: 11, justifyContent: "center", marginBottom: 8 }}>
              <div style={{
                width: 44, height: 44, borderRadius: 11, background: C.green,
                display: "flex", alignItems: "center", justifyContent: "center",
                fontSize: 22, fontWeight: 900, color: C.bg,
              }}>O</div>
              <span style={{ fontSize: 22, fontWeight: 900, letterSpacing: -0.5 }}>OkumuTraders.com</span>
            </div>
            <p style={{ textAlign: "center", color: C.muted, marginBottom: 24, fontSize: 13 }}>
              Sign in to your trading account
            </p>
            <span style={lbl}>EMAIL OR PHONE</span>
            <input value={lPhone} onChange={(e) => setLPhone(e.target.value)} placeholder="e.g. 254712345678" style={{ ...inp(), marginBottom: 16 }} />
            <span style={lbl}>PASSWORD</span>
            <input type="password" value={lPass} onChange={(e) => setLPass(e.target.value)} placeholder="••••••••" style={{ ...inp(), marginBottom: 22 }} />
            <button
              onClick={() => { notify("Welcome back!"); setShowLogin(false); }}
              style={{
                width: "100%", padding: "15px", background: C.green, border: "none",
                borderRadius: 11, fontSize: 16, fontWeight: 800, color: C.bg, cursor: "pointer", marginBottom: 10,
              }}
            >Sign In</button>
            <button onClick={() => setShowLogin(false)} style={{
              width: "100%", padding: "12px", background: "transparent", border: "none",
              fontSize: 14, fontWeight: 700, color: C.muted, cursor: "pointer",
            }}>Cancel</button>
            <p style={{ textAlign: "center", fontSize: 11, color: C.sub, marginTop: 8, lineHeight: 1.6 }}>
              Licensed in The Bahamas · <strong>BHA-0023-1873201</strong>
            </p>
          </div>
        </div>
      )}

      {/* ── DEPOSIT MODAL ───────────────────────────────────────────────────── */}
      {showDeposit && (
        <div style={{
          position: "fixed", inset: 0, background: "rgba(0,0,0,0.8)",
          display: "flex", alignItems: "center", justifyContent: "center",
          zIndex: 100, padding: 20,
        }}>
          <div style={{
            background: C.card, borderRadius: 20, padding: "28px 22px",
            width: "100%", maxWidth: 360, border: `1.5px solid ${C.border}`,
          }}>
            <h3 style={{ textAlign: "center", marginBottom: 22, fontSize: 18, fontWeight: 800, color: C.green }}>
              Deposit via M-Pesa
            </h3>
            <span style={lbl}>AMOUNT (KES)</span>
            <input value={dAmt} onChange={(e) => setDAmt(e.target.value)} style={{ ...inp(), marginBottom: 16 }} />
            <span style={lbl}>M-PESA PHONE NUMBER</span>
            <input value={dPhone} onChange={(e) => setDPhone(e.target.value)} style={{ ...inp(), marginBottom: 20 }} />
            <button onClick={sendSTK} disabled={stkSent} style={{
              width: "100%", padding: "15px",
              background: stkSent ? "#0a6e2a" : C.green,
              border: "none", borderRadius: 11,
              fontSize: 15, fontWeight: 800, color: C.bg,
              cursor: stkSent ? "not-allowed" : "pointer", marginBottom: 10,
            }}>
              {stkSent ? "Sending..." : "Send STK Push"}
            </button>
            <button onClick={() => setShowDeposit(false)} style={{
              width: "100%", padding: "14px", background: "transparent", border: "none",
              fontSize: 15, fontWeight: 700, color: C.muted, cursor: "pointer",
            }}>Cancel</button>
          </div>
        </div>
      )}

      {/* ── TOAST ───────────────────────────────────────────────────────────── */}
      {notif && (
        <div style={{
          position: "fixed", bottom: 82, left: "50%", transform: "translateX(-50%)",
          background: "#1a2d48", border: `1px solid ${C.border}`,
          borderRadius: 10, padding: "12px 20px",
          fontSize: 13, fontWeight: 600, color: C.text,
          zIndex: 200, whiteSpace: "nowrap",
          boxShadow: "0 4px 24px rgba(0,0,0,0.6)",
        }}>
          {notif}
        </div>
      )}
    </div>
  );
}
